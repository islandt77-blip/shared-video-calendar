        start: els.eventStart.value,

        end: els.eventEnd.value,

        color: state.selectedColor,

        type: els.eventType.value,

        notes: els.eventNotes.value.trim()

      };

      if (!entry.title || !entry.date) return;

      const existingIndex = state.events.findIndex(item => item.id === id);

      if (existingIndex >= 0) state.events.splice(existingIndex, 1, entry);

      else state.events.push(entry);

      state.selectedDate = entry.date;

      state.viewDate = startOfMonth(parseDateKey(entry.date));

      closeEventDialog();

      render();

      showNotice("日程已保存");

    }


    function deleteCurrentEvent() {

      const id = els.eventId.value;

      if (!id) return;

      state.events = state.events.filter(item => item.id !== id);

      closeEventDialog();

      render();

      showNotice("日程已删除");

    }


    function handleDocuments() {

      const files = Array.from(els.docUpload.files || []);

      if (!files.length) return;

      const readers = files.map(file => new Promise(resolve => {

        const reader = new FileReader();

        reader.onload = () => resolve({

          id: `doc-${Date.now()}-${Math.random().toString(16).slice(2)}`,

          name: file.name,

          size: file.size,

          type: file.type,

          addedAt: new Date().toISOString(),

          dataUrl: reader.result

        });

        reader.readAsDataURL(file);

      }));

      Promise.all(readers).then(docs => {

        state.documents.push(...docs);

        els.docUpload.value = "";

        render();

        showNotice("文档已上传到本机浏览器");

      });

    }


    function copyShareLink() {

      const payload = {

        events: state.events,

        colorLabels: state.colorLabels,

        selectedDate: state.selectedDate,

        viewDate: toDateKey(state.viewDate)

      };

      const encoded = btoa(unescape(encodeURIComponent(JSON.stringify(payload))));

      const url = `${location.origin}${location.pathname}#data=${encoded}`;

      navigator.clipboard?.writeText(url).then(() => {

        showNotice("共享链接已复制。部署到公网后，这个链接可被他人打开并继续编辑。");

      }).catch(() => {

        prompt("复制这个共享链接：", url);

      });

    }


    function exportJson() {

      const payload = {

        exportedAt: new Date().toISOString(),

        events: state.events,

        colorLabels: state.colorLabels,

        documents: state.documents.map(({ dataUrl, ...doc }) => doc)

      };

      const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });

      const link = document.createElement("a");

      link.href = URL.createObjectURL(blob);

      link.download = "mac-calendar-backup.json";

      link.click();

      URL.revokeObjectURL(link.href);

      showNotice("备份已导出");

    }


    function importJson() {

      const file = els.importFile.files?.[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = () => {

        try {

          const data = JSON.parse(reader.result);

          if (!Array.isArray(data.events)) throw new Error("Invalid events");

          state.events = normalizeEvents(data.events);

          if (data.colorLabels) state.colorLabels = normalizeColorLabels(data.colorLabels);

          render();

          showNotice("日程已导入");

        } catch (error) {

          showNotice("导入失败，请检查 JSON 文件");

        } finally {

          els.importFile.value = "";

        }

      };

      reader.readAsText(file);

    }


    function loadEvents() {

      try {

        const stored = localStorage.getItem(STORAGE_KEY);

        if (!stored) return [...defaultEvents];

        const parsed = JSON.parse(stored);

        return normalizeEvents(parsed);

      } catch (error) {

        return [...defaultEvents];

      }

    }


    function loadDocuments() {

      try {

        return JSON.parse(localStorage.getItem(DOC_KEY)) || [];

      } catch (error) {

        return [];

      }

    }


    function loadColorLabels() {

      try {

        return normalizeColorLabels(JSON.parse(localStorage.getItem(COLOR_LABEL_KEY)) || {});

      } catch (error) {

        return { ...defaultColorLabels };

      }

    }


    function loadFromShareHash() {

      const match = location.hash.match(/data=([^&]+)/);

      if (!match) return;

      try {

        const payload = JSON.parse(decodeURIComponent(escape(atob(match[1]))));

        if (Array.isArray(payload.events)) state.events = normalizeEvents(payload.events);

        if (payload.colorLabels) state.colorLabels = normalizeColorLabels(payload.colorLabels);

        if (payload.selectedDate) state.selectedDate = payload.selectedDate;

        if (payload.viewDate) state.viewDate = startOfMonth(parseDateKey(payload.viewDate));

        saveEvents();

        showNotice("已从共享链接载入日程");

      } catch (error) {

        showNotice("共享链接数据无法读取");

      }

    }


    function saveEvents() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(state.events));

    }


    function saveDocuments() {

      try {

        localStorage.setItem(DOC_KEY, JSON.stringify(state.documents));

      } catch (error) {

        showNotice("文档较大，本机浏览器空间可能不足");

      }

    }


    function saveColorLabels() {

      localStorage.setItem(COLOR_LABEL_KEY, JSON.stringify(state.colorLabels));

    }


    function normalizeColorLabels(labels) {

      const next = { ...defaultColorLabels };

      colors.forEach(color => {

        if (labels && typeof labels[color] === "string" && labels[color].trim()) {

          next[color] = labels[color].trim();

        }

      });

      return next;

    }


    function colorMeaning(color) {

      return state.colorLabels[color] || defaultColorLabels[color] || "未命名";

    }


    function normalizeEvents(events) {

      return events.map(item => ({

        id: String(item.id || `event-${Date.now()}-${Math.random().toString(16).slice(2)}`),

        title: String(item.title || "未命名日程"),

        date: item.date || toDateKey(new Date()),

        start: item.start || "",

        end: item.end || "",

        color: colors.includes(item.color) ? item.color : (item.color || colors[0]),

        type: item.type === "node" ? "node" : "schedule",

        notes: item.notes || ""

      }));

    }


    function eventsForDate(dateKey) {

      return state.events

        .filter(item => item.date === dateKey)

        .sort((a, b) => (a.start || "99:99").localeCompare(b.start || "99:99"));

    }


    function changeMonth(delta) {

      state.viewDate = new Date(state.viewDate.getFullYear(), state.viewDate.getMonth() + delta, 1);

      render();

    }


    function toDateKey(date) {

      const year = date.getFullYear();

      const month = String(date.getMonth() + 1).padStart(2, "0");

      const day = String(date.getDate()).padStart(2, "0");

      return `${year}-${month}-${day}`;

    }


    function parseDateKey(key) {

      const [year, month, day] = key.split("-").map(Number);

      return new Date(year, month - 1, day);

    }


    function startOfMonth(date) {

      return new Date(date.getFullYear(), date.getMonth(), 1);

    }


    function addDays(date, days) {

      const copy = new Date(date);

      copy.setDate(copy.getDate() + days);

      return copy;

    }


    function formatShortDate(date) {

      return `${date.getMonth() + 1}/${date.getDate()}`;

    }


    function formatFullDate(date) {

      return date.toLocaleDateString("zh-CN", { year: "numeric", month: "long", day: "numeric" });

    }


    function formatTimeRange(item) {

      if (item.start && item.end) return `${item.start}-${item.end}`;

      if (item.start) return item.start;

      return "";

    }


    function formatBytes(bytes) {

      if (bytes < 1024) return `${bytes} B`;

      if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;

      return `${(bytes / 1024 / 1024).toFixed(1)} MB`;

    }


    function escapeHtml(value) {

      return String(value)

        .replaceAll("&", "&amp;")

        .replaceAll("<", "&lt;")

        .replaceAll(">", "&gt;")

        .replaceAll('"', "&quot;")

        .replaceAll("'", "&#039;");

    }

      els.notice.classList.add("show");

      clearTimeout(noticeTimer);
